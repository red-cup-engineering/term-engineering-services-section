# price-predicate-identification
## Requirements
- `cheapest-identification-path`: Given declarations that one predicate is basically another (kind mld) or equal to another (kind equal), the operation returns the cheapest chain of declarations from one predicate to another, its total price, and the chain's kind: equal only when every step is equal, otherwise mld.
- `landauer-pricing`: A step of kind mld is priced 1 (one witness that the two predicates' local invariants agree; the gauge flip itself is reversible and free) and a step of kind equal is priced 2 (the witness plus one irreversible identification bit that erases the distinction); prices add along a chain, and a predicate reaches itself at price 0 with kind equal.
- `unreachable-datum`: When no chain exists the operation returns the same record with path null, price null, and kind null rather than throwing.
## Prohibitions
- `malformed-request-refusal`: The operation must refuse a request whose from or to is not a non-empty string, whose declarations is not an array, or that holds a declaration whose kind is not mld or equal.

## Operation
```mjs path=scripts/price-predicate-identification.mjs
#!/usr/bin/env node
const PRICE = Object.freeze({ mld: 1, equal: 2 });

/**
 * Prices the cheapest chain of mld or equal declarations identifying one predicate with another.
 * @param {object} request A record with declarations ([{a, kind, b}]), from, and to.
 * @returns {object} A PredicateIdentificationPrice record with path, price, and kind (null when unreachable).
 * @throws {TypeError} When from, to, or declarations are malformed or a declaration kind is unknown.
 */
export function pricePredicateIdentification(request) {
  if (typeof request?.from !== "string" || request.from.length === 0 || typeof request?.to !== "string" || request.to.length === 0 || !Array.isArray(request?.declarations)) {
    throw new TypeError("from, to, and declarations are required");
  }
  const edges = new Map();
  const add = (a, kind, b) => { if (!edges.has(a)) edges.set(a, []); edges.get(a).push({ to: b, kind }); };
  for (const declaration of request.declarations) {
    if (typeof declaration?.a !== "string" || typeof declaration?.b !== "string" || PRICE[declaration?.kind] === undefined) throw new TypeError("a declaration needs a, b, and a kind of mld or equal");
    add(declaration.a, declaration.kind, declaration.b);
    add(declaration.b, declaration.kind, declaration.a);
  }
  const best = new Map([[request.from, { price: 0, path: [], allEqual: true }]]);
  const queue = [request.from];
  while (queue.length > 0) {
    queue.sort((x, y) => best.get(x).price - best.get(y).price);
    const current = queue.shift();
    const state = best.get(current);
    for (const edge of edges.get(current) ?? []) {
      const price = state.price + PRICE[edge.kind];
      const known = best.get(edge.to);
      if (known === undefined || price < known.price) {
        best.set(edge.to, { price, path: [...state.path, { a: current, kind: edge.kind, b: edge.to }], allEqual: state.allEqual && edge.kind === "equal" });
        queue.push(edge.to);
      }
    }
  }
  const found = best.get(request.to);
  return {
    type: "PredicateIdentificationPrice",
    from: request.from,
    to: request.to,
    path: found === undefined ? null : found.path,
    price: found === undefined ? null : found.price,
    kind: found === undefined ? null : (found.allEqual ? "equal" : "mld"),
  };
}

if (import.meta.main) {
  if (process.argv.includes("--help")) {
    process.stdout.write("pricePredicateIdentification — Prices the cheapest chain of mld or equal declarations identifying one predicate with another.\n");
  } else {
    let text = "";
    for await (const chunk of process.stdin) text += chunk;
    for (const line of text.split("\n")) {
      if (line.length > 0) process.stdout.write(`${JSON.stringify(pricePredicateIdentification(JSON.parse(line)))}\n`);
    }
  }
}
```
## Tests
```mjs path=test/price-predicate-identification.test.mjs
import assert from "node:assert/strict";
import test from "node:test";
import { pricePredicateIdentification } from "../scripts/price-predicate-identification.mjs";
const declarations = [{ a: "part-of", kind: "mld", b: "component-of" }, { a: "component-of", kind: "equal", b: "constituent-of" }, { a: "cites", kind: "equal", b: "references" }];
test("prices a mixed chain as mld with added prices", () => {
  const out = pricePredicateIdentification({ declarations, from: "part-of", to: "constituent-of" });
  assert.equal(out.price, 3);
  assert.equal(out.kind, "mld");
  assert.equal(out.path.length, 2);
});
test("prices an all-equal chain as equal and is symmetric", () => {
  assert.deepEqual(pricePredicateIdentification({ declarations, from: "references", to: "cites" }), { type: "PredicateIdentificationPrice", from: "references", to: "cites", path: [{ a: "references", kind: "equal", b: "cites" }], price: 2, kind: "equal" });
});
test("a predicate reaches itself at price 0 as equal, and an unrelated one is unreachable", () => {
  assert.equal(pricePredicateIdentification({ declarations, from: "cites", to: "cites" }).price, 0);
  const none = pricePredicateIdentification({ declarations, from: "cites", to: "part-of" });
  assert.equal(none.price, null); assert.equal(none.path, null); assert.equal(none.kind, null);
});
test("refuses an unknown kind", () => assert.throws(() => pricePredicateIdentification({ declarations: [{ a: "x", kind: "same", b: "y" }], from: "x", to: "y" }), /kind of mld or equal/u));
```
